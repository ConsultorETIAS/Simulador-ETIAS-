function etiaswidgetabrir() {
 var q = document.getelementbyid("etias-widget-q").value.trim();
 if (!q) return;
 var base = "https://consultoretias.github.io/simulador-etias-/";
 window.location.href = base + "?q=" + encodeuricomponent(q);
}
